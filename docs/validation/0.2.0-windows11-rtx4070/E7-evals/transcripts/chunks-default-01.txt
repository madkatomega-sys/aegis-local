```python
def chunks(items, size):
    """Split items into consecutive lists of `size` items; the last list may be shorter."""
    result = []
    current = []
    for item in items:
        current.append(item)
        if len(current) >= size:
            result.append(current)
            current = []
    if current:
        result.append(current)
    return result
```
